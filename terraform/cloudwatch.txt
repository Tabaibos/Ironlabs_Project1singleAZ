#### IAM role for cloudwatch agent
######## Creating the CloudWatch Agent Configuration


resource "aws_cloudwatch_log_group" "demo_log_group" {
  name = "joaquim/cloudwatch/logs"
  retention_in_days = 7 # tells us how long we want to keep log events in the specified log group.

  tags = {
    Name = "project1-joaquim-logs"
  }
}

##############################33 IAM role for cloudwatch agent


resource "aws_iam_role" "cloudwatch_agent_role" {
  name = "CloudWatchAgentRole-Joaquim"

  assume_role_policy = jsonencode({ 
    Version = "2012-10-17"
    Statement = [
      {
        Action = "sts:AssumeRole"
        Effect = "Allow"
        Principal = {
          Service = "ec2.amazonaws.com"
        }
      }
    ]
  })
}

resource "aws_iam_role_policy_attachment" "cloudwatch_agent_policy_attachment" {
  policy_arn = "arn:aws:iam::aws:policy/CloudWatchAgentServerPolicy"
  role       = aws_iam_role.cloudwatch_agent_role.name
}

#########Creating the CloudWatch Agent Configuration 

data "template_file" "cloudwatch_agent_config" {
  template = file("${path.module}/cloudwatch_agent_config.json.tpl")

  vars = {
    log_group_name = aws_cloudwatch_log_group.demo_log_group.name
  }
}


resource "aws_ssm_parameter" "cloudwatch_agent_config" {
  name  = "/cloudwatch-agent/config"
  type  = "String"
  value = data.template_file.cloudwatch_agent_config.rendered
}

resource "aws_iam_instance_profile" "cloudwatch_agent_profile" {
  name = "CloudWatchAgentProfile"
  role = aws_iam_role.cloudwatch_agent_role.name
}
